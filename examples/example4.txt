
select
        ps.ps_partkey, 
        avg (ps.ps_supplycost)
from
        part as p,
        partsupp as ps,
        supplier as s
where
        (p.p_partkey = ps.ps_partkey)
        and (s.s_suppkey = ps.ps_suppkey)
group by
        ps.ps_partkey;
